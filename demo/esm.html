This page loads the component class from esm.sh and registers it manually.

Import and register the class

<script type="module">
  import { ScriptDependentElement } from
    'https://esm.sh/@aarongustafson/script-dependent@0.0.1/script-dependent.js?bundle&target=es2022';

  if (!customElements.get('script-dependent')) {
    customElements.define('script-dependent', ScriptDependentElement);
  }
</script>
